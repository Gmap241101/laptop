# Phase 4 Vercel Staging environment-variable reference.
# Configure these in the Vercel gh-pages-3 staging project UI.
# Do NOT create or package .env* files for deploy.ps1 v13.1.

# Required to enable the Phase 4 Clerk diagnostics bridge.
VITE_CLERK_STAGING_ENABLED=true

# Heroku Staging API origin, with no trailing path required.
VITE_API_URL=https://YOUR-HEROKU-STAGING-APP.herokuapp.com

# Clerk Development instance Publishable Key. Must begin with pk_test_.
VITE_CLERK_PUBLISHABLE_KEY=pk_test_REPLACE_ME

# The diagnostics UI is additionally gated by the query string:
#   ?clerkTest=1
# Example:
#   https://YOUR-STAGING-PROJECT.vercel.app/?clerkTest=1
# The dedicated Vercel test project may build with Vite production mode.
# Activation is controlled by VITE_CLERK_STAGING_ENABLED=true, and the UI still requires ?clerkTest=1.


# Phase 9 opt-in member-profile read cutover. Keep this only on the dedicated mkrental staging project.
VITE_MEMBER_PROFILE_POSTGRES_READ_ENABLED=true

# The actual cutover still requires the explicit URL query parameter:
#   ?memberRead=postgres
# Recommended diagnostics URL:
#   https://YOUR-STAGING-PROJECT.vercel.app/?clerkTest=1&memberRead=postgres

# Phase 10 opt-in: disable only the userAccounts/{uid} realtime watcher in the dedicated staging project.
VITE_MEMBER_PROFILE_FIRESTORE_WATCHER_DISABLED=true

# The watcher is disabled only when all Phase 9/10 gates are present:
#   ?memberRead=postgres&memberWatcher=off
# Recommended Phase 10 diagnostics URL:
#   https://YOUR-STAGING-PROJECT.vercel.app/?clerkTest=1&memberRead=postgres&memberWatcher=off
