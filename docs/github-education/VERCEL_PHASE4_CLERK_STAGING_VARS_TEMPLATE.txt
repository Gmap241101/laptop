# Heroku Staging API origin, with no trailing path required.

#   ?clerkTest=1

# The dedicated Vercel test project may build with Vite production mode.
# Activation is controlled by VITE_CLERK_STAGING_ENABLED=true, and the UI still requires ?clerkTest=1.


# Phase 9 opt-in member-profile read cutover. Keep this only on the dedicated mkrental staging project.
VITE_MEMBER_PROFILE_POSTGRES_READ_ENABLED=true

# The actual cutover still requires the explicit URL query parameter:
#   ?memberRead=postgres
# Recommended diagnostics URL:
#   https://YOUR-STAGING-PROJECT.vercel.app/?clerkTest=1&memberRead=postgres

# Phase 10 opt-in: disable only the userAccounts/{uid} realtime watcher in the dedicated staging project.
VITE_MEMBER_PROFILE_FIRESTORE_WATCHER_DISABLED=true

# The watcher is disabled only when all Phase 9/10 gates are present:
#   ?memberRead=postgres&memberWatcher=off
# Recommended Phase 10 diagnostics URL:
#   https://YOUR-STAGING-PROJECT.vercel.app/?clerkTest=1&memberRead=postgres&memberWatcher=off

# Phase 11 opt-in: after a Firestore userAccounts write succeeds, refresh the linked PostgreSQL member shadow.
VITE_MEMBER_PROFILE_WRITE_THROUGH_ENABLED=true

# Start the Phase 11 staging session with all gates present once:
#   ?clerkTest=1&memberRead=postgres&memberWatcher=off&memberWriteThrough=on
# The Phase 9/10/11 test gates are latched in sessionStorage for the current browser tab so SPA navigation
# to /mypage or /admin does not silently disable the test. Close the tab, or use ?memberRead=firestore,
# to reset the PostgreSQL/watcher cutover latch.

# Phase 12 opt-in: serve the current user's rentalRestrictions/{uid} state from PostgreSQL
# and disable that single Firestore realtime listener in the dedicated staging project.
VITE_RENTAL_RESTRICTION_POSTGRES_READ_ENABLED=true

# Start the Phase 12 staging session with:
#   ?clerkTest=1&memberRead=postgres&memberWatcher=off&memberWriteThrough=on&restrictionRead=postgres&restrictionWatcher=off
# restrictionRead=postgres + restrictionWatcher=off is latched in sessionStorage for the current tab.
# Use ?restrictionRead=firestore or close the tab to return this restriction read to Firestore realtime mode.

# Phase 14 opt-in: enable rentalRequests PostgreSQL shadow parallel-parity diagnostics.
# This does NOT cut over application reads or writes. Firestore remains authoritative.
VITE_RENTAL_REQUEST_POSTGRES_PARITY_ENABLED=true

# Start the Phase 14 staging session with:
#   ?clerkTest=1&memberRead=postgres&memberWatcher=off&memberWriteThrough=on&restrictionRead=postgres&restrictionWatcher=off&rentalRequestParity=1
# rentalRequestParity=1 is latched in sessionStorage for the current tab.
# Use ?rentalRequestParity=0 or close the tab to disable Phase 14 parity observation.

# Phase 15 opt-in: use the validated PostgreSQL rental request shadow as the preferred
# user request-history read source, with Firestore fallback.
VITE_RENTAL_REQUEST_POSTGRES_READ_ENABLED=true

# Phase 15 watcher reduction gate. This only takes effect when rentalRequestRead=postgres
# and rentalRequestWatcher=off are explicitly requested in the staging session.
VITE_RENTAL_REQUEST_FIRESTORE_WATCHER_DISABLED=true

# Phase 15 Stage A:
#   ?clerkTest=1&memberRead=postgres&memberWatcher=off&memberWriteThrough=on&restrictionRead=postgres&restrictionWatcher=off&rentalRequestParity=1&rentalRequestRead=postgres
# Phase 15 Stage B after Stage A parity passes:
#   ?clerkTest=1&memberRead=postgres&memberWatcher=off&memberWriteThrough=on&restrictionRead=postgres&restrictionWatcher=off&rentalRequestRead=postgres&rentalRequestWatcher=off
# Use ?rentalRequestRead=firestore&rentalRequestParity=0 or close the tab to reset.
